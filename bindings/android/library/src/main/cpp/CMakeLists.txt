cmake_minimum_required(VERSION 3.22.1)
project(peerloom_android LANGUAGES CXX)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(RUST_TRIPLE "aarch64-linux-android")
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(RUST_TRIPLE "x86_64-linux-android")
else()
  message(FATAL_ERROR "Unvalidated Android ABI: ${ANDROID_ABI}")
endif()
add_library(peerloom_rust STATIC IMPORTED)
set_target_properties(peerloom_rust PROPERTIES IMPORTED_LOCATION "${PEERLOOM_RUST_TARGET_DIR}/${RUST_TRIPLE}/release/libdevice_ffi.a")
add_library(peerloom_android SHARED peerloom_jni.cpp)
target_include_directories(peerloom_android PRIVATE "${PEERLOOM_SDK_ROOT}/device-ffi/include")
target_compile_features(peerloom_android PRIVATE cxx_std_17)
target_link_options(peerloom_android PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--exclude-libs,ALL")
# Exactly one Rust static image contains both the C ABI and Android verifier singleton.
target_link_libraries(peerloom_android PRIVATE peerloom_rust android log dl m)
